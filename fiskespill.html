import turtle import random screen = turtle.Screen() screen.bgcolor("blue") # Fisk fish = turtle.Turtle() fish.shape("circle") fish.color("yellow") fish.penup() fish.speed(0) fish.setheading(90) # Liste for å holde på alle boblene bubbles = [] # Lag 5 bobler for i in range(5): b = turtle.Turtle() b.shape("circle") b.color("white") b.penup() b.speed(0) b.goto(random.randint(-170, 170), random.randint(-170, 170)) bubbles.append(b) # Poeng og vinner-grense score = 0 MAX_SCORE = 10 # Bruker store bokstaver for å matche sjekken i koden p = turtle.Turtle() p.penup() p.color("black") p.hideturtle() p.goto(100, 170) p.write("Poeng: 0", font=("Arial", 14)) # Fart move_speed = 15 turn_speed = 15 def forward(): fish.forward(move_speed) check_collision() check_bounds() def backward(): fish.backward(move_speed) check_collision() check_bounds() def left(): fish.left(turn_speed) check_collision() # Rettet: La til () check_bounds() def right(): fish.right(turn_speed) check_collision() # Rettet: La til () check_bounds() # Rettet: La til () def check_collision(): global score # Går igjennom hver eneste boble i listen for å sjekke om fisken treffer den for b in bubbles: if fish.distance(b) < 20: score += 1 p.clear() p.goto(100, 170) # Sikrer at poengsummen tegnes på riktig sted p.write("Poeng: " + str(score), font=("Arial", 14)) # Flytt den spesifikke boblen du traff til et nytt sted b.goto(random.randint(-180, 180), random.randint(-180, 180)) # Sjekk om spilleren har vunnet if score >= MAX_SCORE: p.goto(-90, 0) p.write("Gratulerer! Du vant! 🏆", font=("Arial", 16, "bold")) # Avslutter spillet etter en kort pause så man rekker å se teksten screen.ontimer(screen.bye, 2000) def check_bounds(): x, y = fish.xcor(), fish.ycor() if x > 200: fish.setx(200) if x < -200: fish.setx(-200) if y > 200: fish.sety(200) if y < -200: fish.sety(-200) screen.listen() # La til stor 'L' og 'R' for piltastene, da turtle krever "Left" og "Right" screen.onkey(forward, "Up") screen.onkey(backward, "Down") screen.onkey(left, "Left") screen.onkey(right, "Right") turtle.mainloop()
